 18def collate_fn(batch):
 19    """
 20    Custom collate function for DataLoader to batch images, tokenized questions, answers, 
 21    and additional metadata like question types and subtypes.
 22    
 23    Parameters:
 24    -----------
 25    batch : list
 26        A list of tuples (image, tokenized_question, encoded_answer, question_type, question_subtype).
 27    
 28    Returns:
 29    --------
 30    tuple
 31        A tuple containing:
 32        - images (Tensor): Batch of images.
 33        - questions (Tensor): Padded batch of tokenized questions.
 34        - binary_questions: Batch of binary questions. 
 35        - answers (Tensor): Batch of encoded answers.
 36        - question_types (list): List of question types ('relational' or 'non-relational').
 37        - question_subtypes (list): List of question subtypes (e.g., 'topbottom', 'closest').
 38    """
 39    images, questions, binary_questions, answers, question_types, question_subtypes, image_description_matrices = zip(*batch)
 40
 41    images = torch.stack(images)
 42    questions = [torch.tensor(q, dtype=torch.long) for q in questions]
 43    questions = pad_sequence(questions, batch_first=True, padding_value=0)
 44    binary_questions = torch.stack(binary_questions)
 45    answers = torch.tensor(answers, dtype=torch.long)
 46    image_description_matrices = torch.stack(image_description_matrices)
 47
 48    return images, questions, binary_questions, answers, list(question_types), list(question_subtypes), image_description_matrices

 91            Proportion of the data to be used for training.

 95            Proportion of the data to be used for testing.
 96        """

 98        if not abs(total_size - 1.0) < 1e-6:

106        train_val_paths, test_paths = train_test_split(img_paths, test_size=test_size, random_state=self.random_seed)
107        val_relative_size = val_size / (train_size + val_size)
108        train_paths, val_paths = train_test_split(train_val_paths, test_size=val_relative_size, random_state=self.random_seed)
109
110        self.train_samples = self._generate_samples(data, train_paths)
111        self.val_samples = self._generate_samples(data, val_paths)

119            log_and_print('Test samples: ' + str(len(self.test_samples)), f)

121        # save a random subset of test samples to a text file for establishing human baseline
122        test_subset_size = 200

171    def compute_answer_distribution(self, samples):
172        """
173        Computes the distribution of answers in a set of samples.
174        
175        Parameters:
176        -----------
177        samples : list
178            A list of samples where each sample is a tuple (image_path, question, answer, question_vector).
179        
180        Returns:
181        --------
182        dict
183            An OrderedDict with answers as keys and their normalized weights as values, sorted by weight in descending order.
184        """
185        answer_counts = {}
186        for _, _, answer, _, _ in samples:
187            if answer in answer_counts:
188                answer_counts[answer] += 1
189            else:
190                answer_counts[answer] = 1
191        
192        total_count = len(samples)
193        answer_weights = {answer: count / total_count for answer, count in answer_counts.items()}
194        sorted_weights = OrderedDict(sorted(answer_weights.items(), key=lambda x: x[1], reverse=True))
195        
196        return sorted_weights

221        """
222        @public
223
224        Builds a vocabulary from the dataset questions and assigns an index to each unique word.
225
226        Parameters:
227        -----------

229            A list of samples where each sample contains a question.

234        --------

238        word_counter = Counter()

244        for word, freq in word_counter.items():

328    def __getitem__(self, idx):
329        """
330        Fetch an image and its corresponding tokenized question-answer pair, applying the appropriate transformations.
331
332        Returns:
333        --------
334        tuple
335            A tuple (image, tokenized_question, encoded_answer, question_type, question_subtype, image_description_matrix) where:
336            - image: Transformed image tensor.
337            - tokenized_question: List of token indices.
338            - binary_question: Binary vector tensor representing the question.
339            - encoded_answer: Integer index.
340            - question_type: 'relational' or 'non-relational'.
341            - question_subtype: Specific subtype of the question.
342            - image_description_matrix: Factored representation of the image.
343        """

360    def _process_image_description_matrix_to_tensor(self, image_description_matrix):
361        """
362        Processes a raw state description matrix for a single sample into a 2D tensor.
363
364        Performs one-hot encoding for categorical features and normalization
365        for coordinate features.
366
367        Supports only paper-specific settings not all of the options of d2r2 data generator. 
368
369        Parameters:
370        -----------
371        image_description_matrix : list[list or tuple]
372            The raw state description data for one sample, where each inner
373            list/tuple represents an object's features.
374
375        Returns:
376        --------
377        torch.Tensor
378            A 2D tensor of shape (num_objects, feature_dim) containing the
379            processed object representations.
380        """
381
382        colors = ['red', 'green', 'blue', 'orange', 'pink', 'yellow']
383        shapes = ['circle', 'square']
384        num_colors = len(colors)
385        num_shapes = len(shapes)
386
387        processed_objects = []
388
389        for raw_object_features in image_description_matrix:
390            color_name = raw_object_features[0]
391            shape_name = raw_object_features[1]
392            x_min = raw_object_features[2]
393            y_min = raw_object_features[3]
394            x_max = raw_object_features[4]
395            y_max = raw_object_features[5]
396
397            # one-hot encode color
398            color_idx = colors.index(color_name)
399            color_one_hot = [0.0] * num_colors
400            color_one_hot[color_idx] = 1.0
401
402            # one-hot encode shape
403            shape_idx = shapes.index(shape_name)
404            shape_one_hot = [0.0] * num_shapes
405            shape_one_hot[shape_idx] = 1.0
406
407            # normalize coordinates
408            norm_x_min = self._normalize_coord(x_min)
409            norm_y_min = self._normalize_coord(y_min)
410            norm_x_max = self._normalize_coord(x_max)
411            norm_y_max = self._normalize_coord(y_max)
412
413            # combine into final feature vector
414            feature_vector = color_one_hot + shape_one_hot + [norm_x_min, norm_y_min, norm_x_max, norm_y_max]
415            processed_objects.append(feature_vector)
416
417        # expected shape: (6, 12)
418        result = torch.tensor(processed_objects, dtype=torch.float)
419        assert result.size() == (6, 12), f"Tensor shape mismatch! Expected (6, 12), but got {result.size()}"
420        return result
421
422    def _normalize_coord(self, coord, dim_size=75):
423        """Normalizes a single coordinate to the range [-1, 1]."""
424        coord_float = float(coord)
425        return (2.0 * coord_float / dim_size) - 1.0

427    def _tokenize_question(self, question):
428        """
429        @public
430
431        Tokenizes the question into a list of token indices and pads or truncates to max_len.
432
433        Parameters:
434        -----------
435        question : str
436            The question to tokenize.
437
438        Returns:
439        --------
440        list
441            A list of token indices.
442        """
443        tokens = [self.vocab.get(word, self.vocab["<UNK>"]) for word in question.lower().split()]
444        tokens = tokens[:self.max_len]  # truncate if longer than max_len
445        tokens += [self.vocab["<PAD>"]] * (self.max_len - len(tokens))  # pad if shorter than max_len
446        return tokens

448    def _encode_answer(self, answer):
449        """
450        Converts an answer into its corresponding index. If the answer is not in the vocabulary,
451        it falls back to an `<UNK>` token if available, otherwise raises an error.
452
453        Parameters:
454        -----------
455        answer : str
456            The answer to encode.
457
458        Returns:
459        --------
460        int
461            The index of the answer in the answer vocabulary.
462
463        Raises:
464        -------
465        KeyError
466            If the answer is not found in `answer_vocab` and `<UNK>` is not available.
467        """
468        if answer in self.answer_vocab:
469            return self.answer_vocab[answer]
470        else:
471            return self.answer_vocab["<UNK>"]

473class ImageAnswerTransform:
474    """
475    A class that applies both image and text transformations, allowing 
476    them to communicate via a shared state. The state can be used to 
477    pass transformation parameters or results from the image transformation 
478    to the text transformation, allowing for coordinated changes between 
479    the image and the label.
480
481    Attributes:
482    -----------
483    image_transform : callable, optional
484        A transformation function for the image. It should accept a PIL Image 
485        and a state dictionary, then return the transformed PIL Image and the updated state.
486        
487    text_transform : callable, optional
488        A transformation function for the label (string). It should accept a string label 
489        and the shared state, then return the transformed label.
490        
491    state : dict, optional
492        A shared state dictionary used to pass information between the image 
493        and text transformations. If not provided, it is initialized as an empty dictionary.
494
495    Methods:
496    --------
497    __call__(img, label)
498        Applies the image transformation, updates the shared state, and then 
499        applies the text transformation using the updated state.
500
501        Parameters:
502        -----------
503        img : PIL.Image
504            The input image to be transformed.
505            
506        label : str
507            The label associated with the image, which may be transformed 
508            based on the shared state.
509
510        Returns:
511        --------
512        img : PIL.Image
513            The transformed image.
514
515        label : str
516            The transformed label, modified based on the image transformation 
517            and the state.
518    """
519
520    def __init__(self, image_transform=None, text_transform=None, state=None):
521        self.image_transform = image_transform
522        self.text_transform = text_transform
523        self.state = state or {}
524
525    def __call__(self, img, label):
526        if self.image_transform:
527            img, self.state = self.image_transform(img, self.state)
528
529        if self.text_transform:
530            label = self.text_transform(label, self.state)
531
532        return img, label

557            If not provided, it is initialized as an empty dictionary.

587        """
