  8class CNNImageEncoder(nn.Module):
  9    """
 10    CNN-based image encoder for Sort-of-CLEVR dataset as described in the paper
 11    'A simple neural network module for relational reasoning'.
 12    
 13    Uses 4 convolutional layers with 32, 64, 128, and 256 kernels respectively,
 14    with ReLU activations and batch normalization.
 15    
 16    Methods:
 17    --------
 18    forward(image: torch.Tensor) -> torch.Tensor
 19        Processes the input image through the CNN to extract feature maps.
 20    """
 21    def __init__(self, input_channels=3):
 22        super(CNNImageEncoder, self).__init__()
 23        
 24        # 4 convolutional layers as described in the paper
 25        self.conv_layers = nn.Sequential(
 26            nn.Conv2d(input_channels, 32, kernel_size=3, stride=2, padding=1),
 27            nn.BatchNorm2d(32),
 28            nn.ReLU(),
 29            
 30            nn.Conv2d(32, 64, kernel_size=3, stride=2, padding=1),
 31            nn.BatchNorm2d(64),
 32            nn.ReLU(),
 33            
 34            nn.Conv2d(64, 128, kernel_size=3, stride=2, padding=1),
 35            nn.BatchNorm2d(128),
 36            nn.ReLU(),
 37            
 38            nn.Conv2d(128, 256, kernel_size=3, stride=2, padding=1),
 39            nn.BatchNorm2d(256),
 40            nn.ReLU()
 41        )
 42    
 43    def forward(self, image):
 44            """
 45            Forward pass to extract features from the input image and add coordinate information.
 46            
 47            Parameters:
 48            -----------
 49            image : torch.Tensor
 50                Input image tensor of shape (batch_size, channels, height, width).
 51                For Sort-of-CLEVR, typically (batch_size, 3, 75, 75).
 52                
 53            Returns:
 54            --------
 55            objects : torch.Tensor
 56                Output tensor with each spatial location as an "object" with content
 57                features and coordinate information.
 58                Shape: (batch_size, d*d, 256+2) where d*d is the number of cells
 59                in the final feature map.
 60            """
 61            # features 
 62            feature_maps = self.conv_layers(image)
 63            batch_size, channels, height, width = feature_maps.shape
 64            
 65            # tagging with normalized spatial coords 
 66
 67            # 1D tensor [height] -> [height, 1] -> [height, 1] * width 
 68            y_coords = torch.linspace(-1, 1, height).unsqueeze(1).expand(height, width)
 69
 70            # 1D tensor [width] -> [height, width]
 71            x_coords = torch.linspace(-1, 1, width).expand(height, width)
 72            
 73            # reshape coords to match feature map dims
 74            coords = torch.stack((y_coords, x_coords), dim=0)  # [2, height, width]
 75            coords = coords.unsqueeze(0).expand(batch_size, 2, height, width)
 76            
 77            # concat feature maps and coordinates along the channel dimension
 78            feature_maps_with_coords = torch.cat([feature_maps, coords.to(feature_maps.device)], dim=1)
 79            
 80            # [batch_size, channels+2, height, width] -> [batch_size, height*width = objects, channels+2 = features + coords]
 81            # [batch_size, channels+2, height, width] -> [batch_size, height*width, channels+2]
 82            objects = feature_maps_with_coords.permute(0, 2, 3, 1).reshape(batch_size, height*width, channels+2)
 83            
 84            return objects

 86class ResNetImageEncoder(nn.Module):
 87    """
 88    ImageEncoder class using a pre-trained ResNet18 model for feature extraction,
 89    with built-in image preprocessing. Doesn't use spatial coordinates of objects as the CNNImageEncoder.
 90
 91    Methods:
 92    --------
 93    preprocess_image(image: torch.Tensor) -> torch.Tensor
 94        Applies preprocessing to an input image.
 95    forward(image: torch.Tensor) -> torch.Tensor
 96        Passes the preprocessed input image through the CNN to extract feature maps.
 97    """
 98
 99    def __init__(self, return_object_features=False):
100        super(ResNetImageEncoder, self).__init__()
101        self.return_object_features = return_object_features
102        # uses a pre-trained ResNet18 model, excluding the last fully connected layer
103        self.cnn = models.resnet18(weights=ResNet18_Weights.IMAGENET1K_V1)
104        self.cnn = nn.Sequential(*list(self.cnn.children())[:-2])
105
106        self.preprocess = transforms.Compose([
107            transforms.Resize((224, 224)),
108            transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])  # normalize with ImageNet stats
109        ])
110
111    def preprocess_image(self, image: torch.Tensor) -> torch.Tensor:
112        """
113        Preprocesses the input image by resizing and normalizing.
114
115        Parameters:
116        -----------
117        image : torch.Tensor
118            Input image tensor of shape (C, H, W).
119
120        Returns:
121        --------
122        torch.Tensor
123            Preprocessed image tensor of shape (1, 3, 224, 224), with a batch dimension.
124        """
125        preprocessed_image = self.preprocess(image)
126        return preprocessed_image
127
128    def forward(self, image: torch.Tensor) -> torch.Tensor:
129        """
130        Forward pass to extract features from the input image.
131
132        Parameters:
133        -----------
134        image : torch.Tensor
135            Input image in tensor format.
136
137        Returns:
138        --------
139        torch.Tensor
140            - if return_object_features: (B, N_objects, F)
141            - else: (B, 512, 7, 7)
142        """
143        preprocessed_image = self.preprocess_image(image)
144        features = self.cnn(preprocessed_image)  # shape: (B, 512, 7, 7)
145
146        if self.return_object_features:
147            B, C, H, W = features.size()
148            features = features.view(B, C, H * W).permute(0, 2, 1)  # (B, 49, 512)
149        return features

189class LSTMQuestionEncoder(nn.Module):
190    """
191    Module for encoding questions using an LSTM.
192
193
194    Methods:
195    -------
196    forward(questions):
197        Performs a forward pass through the network, returning the final hidden state of the LSTM.
198    """
199
200    def __init__(self, vocab_size, embed_size, hidden_size, num_layers):
201        super(LSTMQuestionEncoder, self).__init__()
202        self.embedding = nn.Embedding(vocab_size, embed_size)
203        self.lstm = nn.LSTM(embed_size, hidden_size, num_layers, batch_first=True)
204
205    def forward(self, questions):
206        """
207        Performs a forward pass through the network.
208
209        Parameters
210        ----------
211        questions : torch.Tensor
212            A batch of tokenized questions represented as tensors.
213
214        Returns
215        -------
216        torch.Tensor
217            The last hidden state of the LSTM for each question in the batch, with shape (batch_size, hidden_size).
218        """
219        
220        embedded = self.embedding(questions)  # shape: (batch_size, seq_len, embed_size)
221        _, (hidden, _) = self.lstm(embedded)  # only take the hidden state output
222
223        # return the last hidden state of the LSTM
224        return hidden[-1]  # shape: (batch_size, hidden_size)

266class RelationNetwork(nn.Module):
267    """
268    Module for relational reasoning, composed of two MLPs (g_theta and f_phi).
269
270    Attributes
271    ----------
272    g_theta : nn.Sequential
273        An MLP that takes concatenated object features and question embeddings to learn relations.
274    f_phi : nn.Sequential
275        A final MLP that takes the summed output of all g_theta outputs to produce a final prediction.
276
277    Methods
278    -------
279    forward(object_features, question_embedding):
280        Performs a forward pass through the network using object features and a question embeddings.
281    """
282
283    def __init__(self, question_dim, num_classes, img_feature_dim):
284        """
285        Initializes the RelationalNetwork with two MLPs: g_theta and f_phi.
286
287        Parameters
288        ----------
289        question_dim : int
290            The dimensionality of the question embedding.
291        num_classes : int
292            The number of output classes for the multi-class classification.        
293        """
294        super(RelationNetwork, self).__init__()
295        
296        self.g_theta = nn.Sequential(
297            nn.Linear(img_feature_dim + question_dim, 2000),
298            nn.ReLU(),
299            nn.Linear(2000, 2000),
300            nn.ReLU(),
301            nn.Linear(2000, 2000),
302            nn.ReLU(),
303            nn.Linear(2000, 2000),
304            nn.ReLU()
305        )
306        self.f_phi = nn.Sequential(
307            nn.Linear(2000, 2000),
308            nn.ReLU(),
309            nn.Linear(2000, 1000),
310            nn.ReLU(),
311            nn.Linear(1000, 500),
312            nn.ReLU(),
313            nn.Linear(500, 100),
314            nn.ReLU(),
315            nn.Linear(100, num_classes)
316        )
317    
318    def forward(self, object_features, question_embedding):
319        """
320        Vectorized forward pass, includes self-relations, aligning with paper.
321
322        Parameters
323        ----------
324        object_features : torch.Tensor
325            A tensor containing the features of objects.
326            Tensor shape: (batch_size, num_objects, feature_dim).
327        question_embedding : torch.Tensor
328            A tensor representing the question embedding.
329            Tensor shape: (batch_size, question_dim).
330
331        Returns
332        -------
333        torch.Tensor
334            Output tensor shape: (batch_size, num_classes).
335        """
336
337        batch_size, num_objects, feature_dim = object_features.size()
338
339        # preparing object pair features
340
341        # we want to create a batch of cubes, where the front-view square 
342        # is formed by a set of objects going from the first row to the bottom one
343        # and copy (broadcast) this single col num_objects times across all cols
344        # and the depth of the cube represents the features
345        obj_i = object_features.unsqueeze(2).repeat(1, 1, num_objects, 1)
346
347        # here we apply the same idea, but now we start with objects spread across the first row and
348        # we want to broadcast to all rows 
349        obj_j = object_features.unsqueeze(1).repeat(1, num_objects, 1, 1)
350
351        # now, let's prepare a full copy of the question embedding for each pair
352        question_embedding_expanded = question_embedding.unsqueeze(1).unsqueeze(2).repeat(1, num_objects, num_objects, 1)
353
354        # and finally to conclude this, concat all the cubes - 
355        # features of obj_i, obj_j and question embedding finally come together 
356        # (batch_size, num_objects, num_objects, 2*feature_dim + question_dim)
357        pair_features = torch.cat([obj_i, obj_j, question_embedding_expanded], dim=3)
358
359        # and flatten pairs for processing: (batch_size * num_objects * num_objects, combined_feature_dim)
360        pair_features = pair_features.view(batch_size * num_objects * num_objects, -1)
361
362        # voila, compute g_theta in one go
363        relations = self.g_theta(pair_features)
364
365        # reshape and sum relations per image: (batch_size, num_objects * num_objects, hidden_dim)
366        relations = relations.view(batch_size, num_objects * num_objects, -1)
367        relations_sum = relations.sum(dim=1)
368
369        # final processing via f_phi
370        output = self.f_phi(relations_sum)
371
372        return output

374class RelationalReasoningModel(nn.Module):
375    """
376    Model for visual question answering using relational reasoning.
377
378    This model consists of three main components:
379    - ImageEncoder: encodes the input image into object features
380    - QuestionEncoder: encodes the input question into a vector embedding
381    - RelationalNetwork: performs relational reasoning over object features and the question embedding
382
383    Methods
384    -------
385    forward(image, question):
386        Performs a forward pass through the network, processing the image and question to produce an output.
387    """
388
389    def __init__(self, img_arch, vocab_size, embed_size, hidden_size, num_layers, num_classes, question_form, image_form):
390        """
391        Initializes the RelationalReasoningModel with an ImageEncoder, QuestionEncoder, and RelationalNetwork.
392
393        Parameters
394        ----------
395        img_arch: str
396            The type of image encoder backbone - values: 'cnn' or 'resnet'. 
397        vocab_size : int
398            The size of the vocabulary used in the QuestionEncoder.
399        embed_size : int
400            The dimensionality of the word embeddings in the QuestionEncoder.
401        hidden_size : int
402            The number of features in the hidden state of the LSTM in the QuestionEncoder.
403        num_layers : int
404            The number of recurrent layers in the LSTM of the QuestionEncoder.
405        num_classes : int
406            The number of output classes for the multi-class classification.
407        """
408        super(RelationalReasoningModel, self).__init__()
409        
410        if image_form == 'image':
411            if img_arch == 'cnn':
412                self.image_encoder = CNNImageEncoder()
413                self.img_feature_dim = 2 * (256 + 2)  # 1 object pair, each object has 256 features +2 for spatial coordinates (x, y)
414            elif img_arch == 'resnet':
415                self.image_encoder = ResNetImageEncoder(return_object_features=True)
416                self.img_feature_dim = 2 * 512
417            else:
418                raise ValueError(f"Unsupported image architecture: {img_arch}")
419        elif image_form == 'matrix':
420            self.image_encoder = FactoredRepresentationEncoder()
421            self.img_feature_dim = 2 * 12
422        else:
423            raise ValueError(f"Unsupported image form: {image_form}")
424        
425        if question_form == 'string':
426            self.question_encoder = LSTMQuestionEncoder(vocab_size, embed_size, hidden_size, num_layers)
427        elif question_form == 'binary':
428            self.question_encoder = BinaryQuestionEncoder()
429            hidden_size = 11
430        else:
431            raise ValueError(f"Unsupported question form: {question_form}")  
432
433        self.relation_network = RelationNetwork(hidden_size, num_classes, self.img_feature_dim)
434    
435    def forward(self, image, question):
436        """
437        Performs a forward pass through the model.
438
439        Parameters
440        ----------
441        image : torch.Tensor
442            A tensor representing the input image of shape (batch_size, channels, height, width).
443        question : torch.Tensor
444            A tensor representing the input question of shape (batch_size, seq_len).
445
446        Returns
447        -------
448        torch.Tensor
449            The output tensor, representing the predicted class logits, with shape (batch_size, num_classes).
450        """
451        object_features = self.image_encoder(image)
452        # flatten the spatial dimensions to produce a set of object features
453        # object_features = image_features.view(image_features.size(0), -1, image_features.size(1))
454        question_embedding = self.question_encoder(question)
455        # perform relational reasoning using the object features and question embedding
456        output = self.relation_network(object_features, question_embedding)
457        
458        return output

460class BaselineModel(nn.Module):
461    """
462    Baseline model that combines a CNN-based image encoder and a question encoder
463    (either LSTM-based or binary) with a standard MLP for classification — without
464    any relational reasoning module. Matches the CNN+MLP baseline from the Sort-of-CLEVR paper.
465    """
466
467    def __init__(self, img_arch, vocab_size, embed_size, hidden_size, num_layers, num_classes, question_form, image_form):
468        super(BaselineModel, self).__init__()
469
470        # image encoder
471        if image_form == 'image':
472            if img_arch == 'cnn':
473                self.image_encoder = CNNImageEncoder()  # output shape: (batch_size, 256, 5, 5)
474                self.img_feature_dim = 5 * 5 * (256 + 2)  # +2 for spatial coordinates (x, y)
475            elif img_arch == 'resnet':
476                self.image_encoder = ResNetImageEncoder(return_object_features=False)  # output shape: (batch_size, 512, 7, 7)
477                self.img_feature_dim = 512 * 7 * 7
478            else:
479                raise ValueError(f"Unsupported image architecture: {img_arch}")
480        elif image_form == 'matrix':
481            self.image_encoder = FactoredRepresentationEncoder() # output shape: (batch_size, 6, 12)
482            self.img_feature_dim = 6 * 12
483        else:
484            raise ValueError(f"Unsupported image form: {image_form}")
485
486        # question encoder
487        if question_form == 'string':
488            self.question_encoder = LSTMQuestionEncoder(vocab_size, embed_size, hidden_size, num_layers)
489        elif question_form == 'binary':
490            self.question_encoder = BinaryQuestionEncoder()
491            hidden_size = 11  # fixed for binary question format
492        else:
493            raise ValueError(f"Unsupported question form: {question_form}")
494
495        # MLP for classification (aligned with paper: 4 layers, 2000 neurons each, ReLU)
496        input_dim = self.img_feature_dim + hidden_size
497        self.classifier = nn.Sequential(
498            nn.Linear(input_dim, 2000),
499            nn.ReLU(),
500            nn.Linear(2000, 2000),
501            nn.ReLU(),
502            nn.Linear(2000, 2000),
503            nn.ReLU(),
504            nn.Linear(2000, num_classes)  # final logits for classification
505        )
506
507    def forward(self, image: torch.Tensor, questions: torch.Tensor) -> torch.Tensor:
508        """
509        Forward pass to get the model's prediction.
510
511        Parameters
512        ----------
513        image : torch.Tensor
514            Input image tensor of shape (batch_size, C, H, W).
515        questions : torch.Tensor
516            Input question tensor (either sequence of word indices or binary vector).
517
518        Returns
519        -------
520        torch.Tensor
521            Output prediction tensor of shape (batch_size, num_classes).
522        """
523
524        # encode image and flatten
525        image_features = self.image_encoder(image)  # shape: (B, C, H, W)
526        image_features = image_features.reshape(image_features.size(0), -1)  # shape: (B, img_feature_dim)
527
528        # encode question
529        question_embedding = self.question_encoder(questions)
530
531        # combine and classify
532        combined = torch.cat((image_features, question_embedding), dim=1)  # shape: (B, img_feature_dim + hidden_size)
533        output = self.classifier(combined)
534        return output
